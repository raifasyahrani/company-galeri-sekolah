===== app\Models\Concerns\UrlGambar.php =====
<?php

namespace App\Models\Concerns;

trait UrlGambar
{
    // Gambar bawaan (public/images/...) dan hasil upload (storage) sama-sama didukung.
    public function urlGambar(string $field = 'gambar'): ?string
    {
        $v = $this->{$field};
        if (! $v) {
            return null;
        }
        return str_starts_with($v, 'images/') ? asset($v) : asset('storage/' . $v);
    }
}

    // Path berawalan "images/" = file bawaan di public/images, selain itu = storage
    private static function urlGambar(?string $path): ?string
    {
        if (!$path) {
            return null;
        }
        $path = str_starts_with($path, 'images/') ? $path : 'storage/' . $path;

        return asset(implode('/', array_map('rawurlencode', explode('/', $path))));
    }

    public function getLogoUrlAttribute(): ?string
    {
        return self::urlGambar($this->gambar);
    }

    public function getSampulUrlAttribute(): ?string
    {
        return self::urlGambar($this->foto_sampul);
    }

    // "Rabu,Sabtu" -> ['Rabu','Sabtu'] (urut Senin..Minggu)
    public function getHariListAttribute(): array
    {
        $list = array_filter(array_map('trim', explode(',', (string) $this->hari_latihan)));

        return array_values(array_intersect(self::HARI, $list));
    }

    // "Jumat" / "Rabu & Sabtu" / "Senin - Sabtu"
    public function getJadwalTeksAttribute(): string
    {
        $h = $this->hari_list;

            count($h) === 0 => '-',
            count($h) === 1 => $h[0],
            count($h) === 2 => $h[0] . ' & ' . $h[1],
            default         => $h[0] . ' - ' . end($h),

    }

===== app\Http\Controllers\AdminCrudController.php =====
<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\Storage;

abstract class AdminCrudController extends Controller
{
    abstract protected function cfg(): array;

    protected function beforeSave(array $data, $item): array
    {
        return $data;
    }

    public function index()
    {
        $c = $this->cfg();
        $items = $c['model']::orderBy($c['order'][0], $c['order'][1])->get();
        return view('admin.crud.index', compact('c', 'items'));
    }

    public function create()
    {
        return view('admin.crud.form', ['c' => $this->cfg(), 'item' => null]);
    }

    public function store(Request $request)
    {
        $c = $this->cfg();
        $data = $request->validate($this->rules());
        $data = $this->simpanGambar($request, $data, null);
        $data = $this->beforeSave($data, null);

        $c['model']::create($data);

        return redirect()->route($c['route'])->with('success', $c['title'] . ' berhasil ditambahkan.');
    }

    public function edit($id)
    {
        $c = $this->cfg();
        $item = $c['model']::findOrFail($id);
        return view('admin.crud.form', compact('c', 'item'));
    }

    public function update(Request $request, $id)
    {
        $c = $this->cfg();
        $item = $c['model']::findOrFail($id);
        $data = $request->validate($this->rules());
        $data = $this->simpanGambar($request, $data, $item);
        $data = $this->beforeSave($data, $item);

        $item->update($data);

        return redirect()->route($c['route'])->with('success', $c['title'] . ' berhasil diperbarui.');
    }

    public function destroy($id)
    {
        $c = $this->cfg();
        $item = $c['model']::findOrFail($id);

        foreach ($c['fields'] as $f) {
            if ($f['type'] === 'image') {
                $this->hapusFile($item->{$f['name']});
            }
        }
        $item->delete();

        return redirect()->route($c['route'])->with('success', $c['title'] . ' berhasil dihapus.');
    }

    private function rules(): array
    {
        $rules = [];
        foreach ($this->cfg()['fields'] as $f) {
            $rules[$f['name']] = $f['rules'];
        }
        return $rules;
    }

    private function simpanGambar(Request $request, array $data, $item): array
    {
        $c = $this->cfg();
        foreach ($c['fields'] as $f) {
            if ($f['type'] !== 'image') {
                continue;
            }
            $n = $f['name'];
            if ($request->hasFile($n)) {
                if ($item) {
                    $this->hapusFile($item->$n);
                }
                $data[$n] = $request->file($n)->store($c['folder'], 'public');
            } else {
                unset($data[$n]); // tidak upload baru = gambar lama dipertahankan
            }
        }
        return $data;
    }

    // Gambar bawaan di public/images tidak pernah dihapus
    private function hapusFile(?string $path): void
    {
        if ($path && ! str_starts_with($path, 'images/')) {
            Storage::disk('public')->delete($path);
        }
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\Ekskul;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Storage;
use Illuminate\Validation\Rule;

class AdminEkskulController extends Controller
{
    // Daftar ekstrakurikuler
    public function index()
    {
        // Kalau tabel masih kosong (pertama kali), 10 ekskul awal dimasukkan otomatis
        Ekskul::isiDataAwalJikaKosong();

        $ekskuls = Ekskul::orderBy('urutan')->orderBy('nama')->get();

        return view('admin.ekskul', compact('ekskuls'));
    }

    // Form tambah
    public function create()
    {
        return view('admin.ekskul_form');
    }

    // Simpan ekskul baru
    public function store(Request $request)
    {
        $data = $request->validate($this->aturan(), $this->pesan());
        $data = $this->siapkan($request, $data);

        Ekskul::create($data);

        return redirect()->route('admin.ekskul')
            ->with('success', 'Ekstrakurikuler berhasil ditambahkan.');
    }

    // Form edit
    public function edit(Ekskul $ekskul)
    {
        return view('admin.ekskul_form', compact('ekskul'));
    }

    // Simpan perubahan
    public function update(Request $request, Ekskul $ekskul)
    {
        $data = $request->validate($this->aturan($ekskul), $this->pesan());
        $data = $this->siapkan($request, $data, $ekskul);

        $ekskul->update($data);

        return redirect()->route('admin.ekskul')
            ->with('success', 'Ekstrakurikuler berhasil diperbarui.');
    }

    // Hapus ekskul (beserta file yang diunggah)
    public function destroy(Ekskul $ekskul)
    {
        $this->hapusFile($ekskul->gambar);
        $this->hapusFile($ekskul->foto_sampul);

        $ekskul->delete();

        return redirect()->route('admin.ekskul')
            ->with('success', 'Ekstrakurikuler berhasil dihapus.');
    }

    // Impor manual 10 ekskul awal (dipakai kalau semua ekskul pernah dihapus)
    // Hanya jalan kalau tabel masih kosong, supaya data tidak tertimpa.
    public function impor()
    {
        if (Ekskul::count() > 0) {
            return redirect()->route('admin.ekskul')
                ->with('error', 'Data ekskul sudah ada, impor dibatalkan supaya data tidak tertimpa.');
        }

        try {
            $jumlah = Ekskul::isiDataAwal();
        } catch (\Throwable $e) {
            return redirect()->route('admin.ekskul')
                ->with('error', 'Impor gagal: ' . $e->getMessage());
        }

        return redirect()->route('admin.ekskul')
            ->with('success', $jumlah . ' ekstrakurikuler dari halaman depan berhasil diimpor.');
    }

    // ---------------------------------------------------------------

    private function aturan(?Ekskul $ekskul = null): array
    {
        $unik = Rule::unique('ekskuls', 'nama');
        if ($ekskul) {
            $unik->ignore($ekskul->id);
        }

        return [
            'nama'           => ['required', 'string', 'max:100', $unik],
            'deskripsi'      => ['nullable', 'string', 'max:1000'],

            'foto_sampul'    => ['nullable', 'image', 'mimes:jpg,jpeg,png,webp', 'max:4096'],
        ];
    }

    // Rapikan data sebelum disimpan + proses upload file
    private function siapkan(Request $request, array $data, ?Ekskul $ekskul = null): array
    {
        // checkbox hari -> "Rabu,Sabtu" (urut sesuai hari dalam seminggu)
        $hari = collect(Ekskul::HARI)
            ->intersect($data['hari_latihan'] ?? [])
            ->implode(',');

        $data['hari_latihan'] = $hari !== '' ? $hari : null;
        $data['urutan'] = $data['urutan'] ?? 0;

        $folder = ['gambar' => 'ekskul/logo', 'foto_sampul' => 'ekskul/sampul'];

        foreach ($folder as $kolom => $tujuan) {
            if ($request->hasFile($kolom)) {
                if ($ekskul) {
                    $this->hapusFile($ekskul->{$kolom});
                }
                $data[$kolom] = $request->file($kolom)->store($tujuan, 'public');
            } else {
                unset($data[$kolom]); // tidak diganti -> pakai file lama
            }
        }

        return $data;
    }

    // Hanya hapus file hasil upload; file bawaan di public/images dibiarkan
    private function hapusFile(?string $path): void
    {
        if ($path && ! str_starts_with($path, 'images/')) {
            Storage::disk('public')->delete($path);
        }
    }

<?php

namespace App\Http\Controllers;

use App\Models\Beranda;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Storage;

class AdminBerandaController extends Controller
{
    public function edit()
    {
        $beranda = Beranda::first();
        return view('admin.beranda.edit', compact('beranda'));
    }

    public function update(Request $request)
    {
        $beranda = Beranda::firstOrFail();

        $data = $request->validate([
            'hero_judul'    => 'required|string|max:255',
            'hero_subjudul' => 'nullable|string|max:255',
            'hero_gambar'   => 'nullable|image|max:2048',
            'tombol_teks'   => 'nullable|string|max:50',
            'tombol_link'   => 'nullable|string|max:255',
            'tentang_judul' => 'required|string|max:255',
            'tentang_isi'   => 'required|string',
            'stat_siswa'    => 'required|integer|min:0',
            'stat_guru'     => 'required|integer|min:0',
            'stat_jurusan'  => 'required|integer|min:0',
            'stat_prestasi' => 'required|integer|min:0',
            'alamat'        => 'nullable|string|max:255',
            'telepon'       => 'nullable|string|max:50',
            'email_sekolah' => 'nullable|email|max:255',
        ]);

        if ($request->hasFile('hero_gambar')) {
            if ($beranda->hero_gambar) {
                Storage::disk('public')->delete($beranda->hero_gambar);
            }
            $data['hero_gambar'] = $request->file('hero_gambar')->store('beranda', 'public');
        } else {
            unset($data['hero_gambar']);
        }

        $beranda->update($data);

        return redirect()->route('admin.beranda')->with('success', 'Beranda berhasil diperbarui.');
    }
}
